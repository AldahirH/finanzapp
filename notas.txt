3.- // 4. Función global: escapa texto antes de mostrarlo en HTML (anti-XSS).
function e(?string $texto): string
{
    return htmlspecialchars($texto ?? '', ENT_QUOTES, 'UTF-8');
}

4.- // 5. Averiguamos qué ruta pidieron y con qué método.
$route = $_GET['r'] ?? 'login';
$method = $_SERVER['REQUEST_METHOD'];

5.- // 7. De la ruta sacamos [clase, acción] y la ejecutamos.
[$controlador, $accion] = $routes[$route][$method];
(new $controlador())->$accion();

6.- // 6. Si no existe esa ruta ni ese método, terminamos con error.
if (!is_string($route) || !isset($routes[$route][$method])) {
    http_response_code(404);
    exit('No encontrado.');
}
